:=============================================================================
:         INPUT PARAMETERS NAMELIST FOR IBM EQUATIONS
:=============================================================================
:
:-----------------------------------------------------------------------------
: namibmin              INPUT   
:-----------------------------------------------------------------------------
:
: file_trajec : name of file with characteristics for each patch
: dir_pathout : base name/path of the result files
: itypepatch  : type of definition of patch initial position:
:              1: circle, 2: rectangle, 3:netcdf
:            choose depending on the file file_trajec you use in inputs directory
:-----------------------------------------------------------------------------
 &namibmin
      file_trajec = 'TEST_CASES/foil_ibm_patch.dat'
      dir_pathout = 'CROCO_FILES/'
      itypepatch = 3 /
:
:-----------------------------------------------------------------------------
: namibmdiff              INPUT
:-----------------------------------------------------------------------------
:
: dtz        : time-step of the random walk on the vertical, for consideration
:                        of vertical diffusion
:        See (Ross and Sharples, 2004, Limnol.Oceanogr.) for sensitivity analysis
:        Few seconds make sure there are no accumulation in low diffusivity areas
:                  and near boundaries
: hdiff            : Horizontal diffusion coefficient
:
:-----------------------------------------------------------------------------
:
 &namibmdiff
      dtz  = 30
      hdiff = 5.0 /
:
:-----------------------------------------------------------------------------
: namibmrestart      SAVE-RESTART CONDITIONS (Not ready yet)
:-----------------------------------------------------------------------------
:
: ibm_restart     : patch is read from restart file
: ibm_time        : use time of restart file
:
:-----------------------------------------------------------------------------
:
 &namibmrestart
      ibm_restart = .false.
      ibm_l_time  = .false. /
:
:-----------------------------------------------------------------------------
: namibmbio      BIO PARAMETERS FOR THE IBM 
:-----------------------------------------------------------------------------
:
: w_max   : maximum vertical velocity (m.s-1)
: alpha_w : factor of convergence for the Hyperbolic tangent function used to
:            weight velocity to a target depth for migration
:-----------------------------------------------------------------------------
:
 &namibmbio
      w_max = 0.0
      alpha_w  = 0.8 /
:
:-----------------------------------------------------------------------------
: namibmpop      OUTPUT PARAMETERS FOR THE IBM
:-----------------------------------------------------------------------------
:
: repro          : activate or deactivate reproduction of species
: dt_spawn       : spawning interval (hours)
: dt_save        : saving interval for new particles
: max_part       : maximum number of particles in NetCDF output file
: duration_ibm_anc  : maximum number of days an anchovy particle can live
: duration_ibm_sar  : maximum number of days a sardine particle can live
: fish_mort         : applying fishing mortality (true or false)
: fishing_strategy  : fishing strategy: 
:                     'historical' for historical time series; 
:                     'HCR' for harvest control rule; 
:                     'Catch' for total allowable catch; 
:                     or F_eval for ... ?
: multiplier_tac    : multiplicative factor to regulate catches, between 0 and 1
: density_dependent : Logical (mortality on larval stage as Somarakis)
:-----------------------------------------------------------------------------
 &namibmpop
        repro          = .false.
        dt_spawn       = 24
        dt_save        = 24
        max_part       = 1500
        duration_ibm_anc  = 1460
        duration_ibm_sar  = 2250
        fish_mort         = .false.
        fishing_strategy  = 'Catch'
        multiplier_tac    = 0.8
        density_dependent = .false./
:
:-----------------------------------------------------------------------------
: namibmdeb      DEB PARAMETERS FOR THE IBM (not ready yet)
:-----------------------------------------------------------------------------
:
:      ! Configuration parameters for the DEB
:      debuse     : logical
:      F_Fix      : logical, if false then read food from a provided file
:      f_fix      : f (f=X/X+K) for DEB if read_environ = False and for initial condition
:      file_NBSS  : file with slope and intercept for NBSS
:      file_food  : file withzoop food from ECO-MARS
:      frac_deb_death : Logical. Fraction of individuals dying within a super individual (.true.) or all (.false.)
:
:-----------------------------------------------------------------------------
:
 &namibmdeb
      debuse    =.true.
      F_Fix     =.true.
      ffix      = 0.8
      file_NBSS =''
      file_food ='' 
      frac_deb_death = .false./
:
:-----------------------------------------------------------------------------
: namibmfrc      DEB FORCING FILE FOR FISH 
:-----------------------------------------------------------------------------
:
:   ! Set file parameters for the DEB
:   fileanchovy      : file with DEB parameters for anchovy
:   filesardine      : file with DEB parameters for sardine
:   catch_anc_bob    : file with monthly anchovy catches in Bay of Biscay 
:   catch_sar_bob    : file with monthly sardine catches in Bay of Biscay
:   fileprobadistrib : file with distribution probability of adults fish
:   nbSizeClass      : number of fish size class in fileprobadistrib
:   sizemin          : Minimal size of fish in fileprobadistrib
:-----------------------------------------------------------------------------
:
 &namibmfrc
        fileanchovy          = 'TEST_CASES/foil_deb_parameters_anc.txt'
        catch_anc_bob        = 'TEST_CASES/foil_catch_anc.txt'
        fileprobadistrib_anc = 'CROCO_FILES/foil_probadistrib.nc'
        nbSizeClass_anc      = 36
        sizemin_anc          = 3.5
        
        filesardine          = 'TEST_CASES/foil_deb_parameters_sar.txt'
        catch_sar_bob        = 'TEST_CASES/foil_catch_sar.txt'
        fileprobadistrib_sar = 'CROCO_FILES/foil_probadistrib.nc'
        nbSizeClass_sar      = 36 
        sizemin_sar          = 3.5
/
