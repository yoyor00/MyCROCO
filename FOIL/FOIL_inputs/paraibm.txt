:=============================================================================
:                 FOIL INDIVIDUAL-BASED MODEL CONFIGURATION
:                 INPUT PARAMETERS GROUPED BY NAMELIST
:=============================================================================
:
:-----------------------------------------------------------------------------
: namibmin          GENERAL INPUT SETTINGS
:-----------------------------------------------------------------------------
:
: file_trajec : name of the patch initialization file
: dir_pathout : directory used for trajectory output and restart files
: itypepatch  : method used to define the initial patch positions:
:               1 = circle, 2 = rectangle, 3 = NetCDF positions
:               This value must match the format of file_trajec.
:
:-----------------------------------------------------------------------------
:
 &namibmin
      file_trajec = 'ibm_patch.dat'
      dir_pathout = '../output_foil/'
      itypepatch = 3 /
:
:-----------------------------------------------------------------------------
: namibmdiff        PARTICLE DIFFUSION SETTINGS
:-----------------------------------------------------------------------------
:
: dtz   : time step (s) used by the vertical random-walk subloop
:         A value of a few seconds helps prevent particle accumulation in
:         low-diffusivity regions and near boundaries. See Ross and Sharples
:         (2004, Limnology and Oceanography) for a sensitivity analysis.
: hdiff : horizontal diffusion coefficient (m2 s-1); set to 0 to disable
:         horizontal diffusion
:
:-----------------------------------------------------------------------------
:
 &namibmdiff
      dtz  = 3
      hdiff = 5.0 /
:
:-----------------------------------------------------------------------------
: namibmrestart     RESTART SETTINGS
:-----------------------------------------------------------------------------
:
: ibm_restart : if true, initialize particle properties from restart files
: ibm_l_time  : if true, resume from the time stored in the restart file;
:               otherwise, use the configured patch start time
:
:-----------------------------------------------------------------------------
:
 &namibmrestart
      ibm_restart = .false.
      ibm_l_time  = .false. /
:
:-----------------------------------------------------------------------------
: namibmbio         VERTICAL-BEHAVIOR SETTINGS
:-----------------------------------------------------------------------------
:
: w_max   : nominal maximum vertical swimming speed (m s-1); the current
:           life-stage routines replace it with stage-specific values
: alpha_w : convergence factor in the hyperbolic-tangent function used to
:           calculate swimming velocity toward the target depth
:-----------------------------------------------------------------------------
:
 &namibmbio
      w_max = 0.0
      alpha_w  = 0.8 /
:
:-----------------------------------------------------------------------------
: namibmpop         POPULATION, REPRODUCTION, AND FISHING SETTINGS
:-----------------------------------------------------------------------------
:
: repro             : enable or disable reproduction
: dt_spawn          : interval (h) between spawning events
: dt_save           : output interval (h) assigned to patches created through
:                     reproduction; initial patches use the interval specified
:                     in file_trajec
: max_part          : maximum number of particles in each offspring patch and
:                     its NetCDF trajectory file
: duration_ibm_anc  : maximum anchovy lifespan (days)
: duration_ibm_sar  : maximum sardine lifespan (days)
: fish_mort         : enable or disable fishing mortality
: fishing_strategy  : fishing-mortality method; accepted values are case-sensitive:
:                     'historical' = prescribed historical mortality,
:                     'HCR' = harvest control rule,
:                     'Catch' = prescribed catch time series,
:                     'F_eval' = fishing-mortality evaluation scenario
: multiplier_tac    : intended TAC scaling factor between 0 and 1; currently
:                     read from this file but not applied by the active code
: density_dependent : intended switch for density-dependent larval mortality;
:                     currently read from this file but not used by the active code
:-----------------------------------------------------------------------------
:
 &namibmpop
        repro             = .true.
        dt_spawn          = 72
        dt_save           = 24
        max_part          = 25000
        duration_ibm_anc  = 1460
        duration_ibm_sar  = 2980
        fish_mort         = .true.
        fishing_strategy  = 'historical'
        multiplier_tac    = 0.8
        density_dependent = .false./
:
:-----------------------------------------------------------------------------
: namibmdeb         DYNAMIC ENERGY BUDGET (DEB) SETTINGS
:-----------------------------------------------------------------------------
:
: debuse        : enable the DEB model and full life cycle; if false, particles
:                 retain the simpler drifting-larvae behavior
: F_Fix         : if true, use the constant functional response ffix; if false,
:                 calculate food availability from file_food
: ffix          : constant DEB functional response, f = X/(X + K), used when
:                 F_Fix is true
: file_NBSS     : NetCDF file containing NBSS slope and intercept fields (from MARS, not linked with CROCO yet)
: file_food     : NetCDF file containing offline zooplankton food fields
: frac_deb_death: if true, DEB starvation removes a fraction of the individuals
:                 represented by a particle; if false, it removes the entire particle
:
:-----------------------------------------------------------------------------
:
 &namibmdeb
      debuse    =.true.
      F_Fix     =.false.
      ffix      = 0.8
      file_NBSS ='../inputs/NBSS_manga.nc'
      file_food ='ibm_zooc.nc' 
      frac_deb_death = .false./
:
:-----------------------------------------------------------------------------
: namibmfrc         EXTERNAL FORCING FILES
:-----------------------------------------------------------------------------
:
: fileanchovy          : anchovy DEB parameter file
: filesardine          : sardine DEB parameter file
: catch_anc_bob        : monthly anchovy catch file for the Bay of Biscay
: catch_sar_bob        : monthly sardine catch file for the Bay of Biscay
: fileprobadistrib_anc : NetCDF file containing anchovy distribution probabilities
: nbSizeClass_anc      : number of anchovy size classes in that file
: sizemin_anc          : minimum anchovy length represented in that file (cm)
: fileprobadistrib_sar : NetCDF file containing sardine distribution probabilities
: nbSizeClass_sar      : number of sardine size classes in that file
: sizemin_sar          : minimum sardine length represented in that file (cm)
:-----------------------------------------------------------------------------
:
 &namibmfrc
        fileanchovy          = '../croco-mustang/FOIL_inputs/deb_parameters_anc.txt'
        filesardine          = '../croco-mustang/FOIL_inputs/deb_parameters_sar.txt'
        catch_anc_bob        = '../inputs/catch_anc_bob.txt'
        catch_sar_bob        = '../inputs/catch_sar_bob.txt'
        fileprobadistrib_anc = '../inputs/probadistrib_1_2.nc'
        nbSizeClass_anc      = 36
        sizemin_anc          = 3.5
        fileprobadistrib_sar = '../inputs/probadistrib_1_2.nc'
        nbSizeClass_sar      = 36
        sizemin_sar          = 3.5/
