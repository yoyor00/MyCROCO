:=============================================================================
:                 FOIL INDIVIDUAL-BASED MODEL CONFIGURATION
:                 INPUT PARAMETERS GROUPED BY NAMELIST
:=============================================================================
:
:-----------------------------------------------------------------------------
: namibmmove         MOVEMENT SETTINGS
:-----------------------------------------------------------------------------
:
: w_max   : nominal maximum vertical swimming speed (m s-1); the current
:           life-stage routines replace it with stage-specific values
: alpha_w : convergence factor in the hyperbolic-tangent function used to
:           calculate swimming velocity toward the target depth
: adult_move : activate (or not) the movement on the horizontal for juv/adults 
:-----------------------------------------------------------------------------
:
 &namibmmove
      w_max = 0.0
      alpha_w  = 0.8 
      adult_move = .true. /
:
:-----------------------------------------------------------------------------
: namibmpop         POPULATION, REPRODUCTION, AND FISHING SETTINGS
:-----------------------------------------------------------------------------
:
: repro             : enable or disable reproduction
: dt_spawn          : interval (h) between spawning events
: max_part          : maximum number of particles in each offspring patch and
:                     its NetCDF trajectory file
: fish_mort         : enable or disable fishing mortality
: fishing_strategy  : fishing-mortality method; accepted values are case-sensitive:
:                     'historical' = prescribed historical mortality,
:                     'HCR' = harvest control rule,
:                     'Catch' = prescribed catch time series,
:                     'F_eval' = fishing-mortality evaluation scenario
: multiplier_tac    : intended TAC scaling factor between 0 and 1; currently
:                     read from this file but not applied by the active code
: density_dependent : intended switch for density-dependent larval mortality;
:                     currently read from this file but not used by the active code
:-----------------------------------------------------------------------------
:
 &namibmpop
        repro          = .true.
        dt_spawn       = 24
        max_part       = 1500
        fish_mort         = .true.
        fishing_strategy  = 'Catch'
        multiplier_tac    = 0.8
        density_dependent = .false. /
:
:-----------------------------------------------------------------------------
: namibmdeb         DYNAMIC ENERGY BUDGET (DEB) SETTINGS
:-----------------------------------------------------------------------------
:
: debuse        : enable the DEB model and full life cycle; if false, particles
:                 retain the simpler drifting-larvae behavior
: F_Fix         : if true, use the constant functional response ffix; if false,
:                 calculate food availability from file_food
: ffix          : constant DEB functional response, f = X/(X + K), used when
:                 F_Fix is true
: file_NBSS     : NetCDF file containing NBSS slope and intercept fields (from MARS, not linked with CROCO yet)
: file_food     : NetCDF file containing offline zooplankton food fields
: frac_deb_death: if true, DEB starvation removes a fraction of the individuals
:                 represented by a particle; if false, it removes the entire particle
:
:-----------------------------------------------------------------------------
:
 &namibmdeb
	debuse    =.true.
	F_Fix     =.true.
	ffix      = 0.8
	file_NBSS ='../inputs/NBSS_manga.nc'
	file_food ='./inputs/init_FOIL/DEB_inputs/cmems_mod_glo_bgc_my_0.083deg-lmtl_P1D-i_1765190183327.nc' 
	frac_deb_death = .false. /
:
:-----------------------------------------------------------------------------
: namibmfiles       SPECIES PARAMETER FILES
:-----------------------------------------------------------------------------
:
: fileanchovy : anchovy parameter file
: filesardine : sardine parameter file
:-----------------------------------------------------------------------------
:
 &namibmfiles
        fileanchovy = './inputs/init_FOIL/parameters_anchovy.txt'
        filesardine = './inputs/init_FOIL/parameters_sardine.txt' /
